imageLoader.js="// imageLoader.js: turns a File into ImageData via an offscreen canvas.
// Images are downscaled before reading pixels — palette accuracy doesn't
// need full resolution, and it keeps large photos fast.

function loadImageToCanvas(file, maxDimension = 200) {
  return new Promise((resolve, reject) => {
    const objectUrl = URL.createObjectURL(file);
    const img = new Image();

    img.onload = () => {
      let { width, height } = img;

      if (width > height && width > maxDimension) {
        height = Math.round(height * (maxDimension / width));
        width = maxDimension;
      } else if (height >= width && height > maxDimension) {
        width = Math.round(width * (maxDimension / height));
        height = maxDimension;
      }

      const canvas = document.createElement('canvas');
      canvas.width = width;
      canvas.height = height;
      const ctx = canvas.getContext('2d', { willReadFrequently: true });
      ctx.drawImage(img, 0, 0, width, height);

      URL.revokeObjectURL(objectUrl);

      try {
        const imageData = ctx.getImageData(0, 0, width, height);
        resolve(imageData);
      } catch (err) {
        // Most commonly a tainted-canvas security error; shouldn't happen
        // for local File objects, but fail gracefully either way.
        reject(err);
      }
    };

    img.onerror = () => {
      URL.revokeObjectURL(objectUrl);
      reject(new Error('Could not load image data.'));
    };

    img.src = objectUrl;
  });
}

window.ImageLoader = { loadImageToCanvas };"

colorUtils.js = "// colorUtils.js: pure conversion helpers, no DOM access.

function rgbToHex(r, g, b) {
  const toHex = (n) => n.toString(16).padStart(2, '0');
  return `#${toHex(r)}${toHex(g)}${toHex(b)}`.toUpperCase();
}

function rgbToHsl(r, g, b) {
  r /= 255; g /= 255; b /= 255;
  const max = Math.max(r, g, b);
  const min = Math.min(r, g, b);
  let h = 0;
  let s = 0;
  const l = (max + min) / 2;

  const delta = max - min;
  if (delta !== 0) {
    s = delta / (1 - Math.abs(2 * l - 1));
    switch (max) {
      case r: h = ((g - b) / delta) % 6; break;
      case g: h = (b - r) / delta + 2; break;
      case b: h = (r - g) / delta + 4; break;
    }
    h *= 60;
    if (h < 0) h += 360;
  }

  return {
    h: Math.round(h),
    s: Math.round(s * 100),
    l: Math.round(l * 100),
  };
}

// Finds the closest entry in NAMED_COLORS by plain Euclidean RGB distance.
// Simple, but with ~100 well-spread reference colors it resolves far more
// distinctly than a 12-hue-bucket scheme, so visually different colors
// (e.g. a tan vs. a brownish gray) land on different names instead of both
// rounding down to the same "Muted Orange".
function nearestColorName(r, g, b) {
  let best = NAMED_COLORS[0];
  let bestDist = Infinity;

  for (const candidate of NAMED_COLORS) {
    const dr = candidate.r - r;
    const dg = candidate.g - g;
    const db = candidate.b - b;
    const dist = dr * dr + dg * dg + db * db;
    if (dist < bestDist) {
      bestDist = dist;
      best = candidate;
    }
  }

  return best.name;
}

colorExtractor.js = "// colorExtractor.js: reduces raw pixel data down to a handful of dominant colors.
//
// Approach: real k-means clustering in RGB space. This costs more CPU than a
// bucket-rounding heuristic, but doesn't suffer from boundary-splitting
// artifacts (near-duplicate colors reported separately because two pixels
// landed a fraction of a value on either side of a rounding threshold).
// K-means finds `count` centroids that genuinely minimize distance to all
// sampled pixels, so visually similar colors converge to one center instead
// of fragmenting.

const ALPHA_THRESHOLD = 128; // ignore mostly-transparent pixels
const MAX_ITERATIONS = 12;
const CONVERGENCE_THRESHOLD = 1; // stop early once centroids move less than this (per channel)

function colorDistanceSq(a, r, g, b) {
  const dr = a.r - r;
  const dg = a.g - g;
  const db = a.b - b;
  return dr * dr + dg * dg + db * db;
}

// k-means++ initialization: pick the first centroid at random, then each
// subsequent one with probability proportional to its squared distance from
// the nearest existing centroid. This spreads the starting points out
// across the color space instead of risking several random picks landing on
// the same dominant color (which plain random init can do).
function pickInitialCentroids(pixels, k) {
  const centroids = [pixels[Math.floor(Math.random() * pixels.length)]];

  while (centroids.length < k) {
    const distances = pixels.map((p) => {
      let minDist = Infinity;
      for (const c of centroids) {
        const d = colorDistanceSq(c, p.r, p.g, p.b);
        if (d < minDist) minDist = d;
      }
      return minDist;
    });

    const totalDist = distances.reduce((sum, d) => sum + d, 0);
    if (totalDist === 0) {
      // All remaining pixels are identical to an existing centroid; pad
      // with random picks to reach k.
      centroids.push(pixels[Math.floor(Math.random() * pixels.length)]);
      continue;
    }

    let threshold = Math.random() * totalDist;
    let chosen = pixels[0];
    for (let i = 0; i < pixels.length; i++) {
      threshold -= distances[i];
      if (threshold <= 0) {
        chosen = pixels[i];
        break;
      }
    }
    centroids.push(chosen);
  }

  return centroids.map((p) => ({ r: p.r, g: p.g, b: p.b }));
}

function kMeans(pixels, k) {
  let centroids = pickInitialCentroids(pixels, k);

  for (let iter = 0; iter < MAX_ITERATIONS; iter++) {
    const sums = centroids.map(() => ({ r: 0, g: 0, b: 0, count: 0 }));

    for (const p of pixels) {
      let bestIndex = 0;
      let bestDist = Infinity;
      for (let i = 0; i < centroids.length; i++) {
        const d = colorDistanceSq(centroids[i], p.r, p.g, p.b);
        if (d < bestDist) {
          bestDist = d;
          bestIndex = i;
        }
      }
      const sum = sums[bestIndex];
      sum.r += p.r;
      sum.g += p.g;
      sum.b += p.b;
      sum.count += 1;
    }

    let maxShift = 0;
    const nextCentroids = centroids.map((centroid, i) => {
      const sum = sums[i];
      if (sum.count === 0) return centroid; // no pixels assigned; keep as-is
      const next = {
        r: sum.r / sum.count,
        g: sum.g / sum.count,
        b: sum.b / sum.count,
        count: sum.count,
      };
      const shift = Math.sqrt(colorDistanceSq(centroid, next.r, next.g, next.b));
      if (shift > maxShift) maxShift = shift;
      return next;
    });

    centroids = nextCentroids;
    if (maxShift < CONVERGENCE_THRESHOLD) break;
  }

  return centroids.filter((c) => c.count > 0);
}

function extractColors(imageData, count = 6) {
  const { data } = imageData;
  const pixels = [];

  for (let i = 0; i < data.length; i += 4) {
    const a = data[i + 3];
    if (a < ALPHA_THRESHOLD) continue;
    pixels.push({ r: data[i], g: data[i + 1], b: data[i + 2] });
  }

  if (pixels.length === 0) return [];

  const k = Math.min(count, pixels.length);
  const clusters = kMeans(pixels, k).sort((a, b) => b.count - a.count);

  return clusters.map((c) => ({
    r: Math.round(c.r),
    g: Math.round(c.g),
    b: Math.round(c.b),
    percentage: (c.count / pixels.length) * 100,
  }));
}

window.ColorExtractor = { extractColors };"
